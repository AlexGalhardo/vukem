```go
app, err := vukem.New(
	vukem.WithCacheStore(rc.CacheStore("cache:")), // also backs locks
	vukem.WithSessionStore(rc.SessionStore("")),   // "session:" keys with TTL
	vukem.WithQueueDriver(rc.Queue(redis.QueueOptions{})),
	// ...
)

```

### Reading `REDIS_URL` from the app configuration

`os.Getenv` above runs before `vukem.New` loads `.env`, `.env.local` and the sources `vukemtest` passes. To read the URL from `app.Source`, use the lazy options: each function runs on the first `app.Cache()`, `app.Sessions()` or `app.Queue()` call, after configuration loaded, and its error is returned by that call (`DefaultMiddleware` builds sessions at boot, so a bad URL is still a boot error).

```go
var (
	once  sync.Once
	rc    *redis.Client
	rcErr error
)

client := func(app *vukem.App) (*redis.Client, error) {
	once.Do(func() {
		if rc, rcErr = redis.Open(context.Background(), app.Source.Get("REDIS_URL")); rcErr != nil {
			return
		}

		app.AddHealthCheck("redis", rc.Ping)
		app.OnShutdown(func(context.Context) error { return rc.Close() })
	})

	return rc, rcErr
}

app, err := vukem.New(
	vukem.WithCacheStoreFunc(func(app *vukem.App) (cache.Store, error) {
		c, err := client(app)
		if err != nil {
			return nil, err
		}

		return c.CacheStore("cache:"), nil
	}),
	vukem.WithSessionStoreFunc(func(app *vukem.App) (session.Store, error) {
		c, err := client(app)
		if err != nil {
			return nil, err
		}

		return c.SessionStore(""), nil
	}),
	vukem.WithQueueDriverFunc(func(app *vukem.App) (queue.Driver, error) {
		c, err := client(app)
		if err != nil {
			return nil, err
		}

		return c.Queue(redis.QueueOptions{}), nil
	}),
)
```

The last of `WithCacheStore` / `WithCacheStoreFunc` (and likewise for sessions and queues) wins; `SetSessionStore` from a provider still overrides both.

| Feature | API |
| --- | --- |
| Cache store | `rc.CacheStore(prefix)` implements `cache.Store`; `Flush` deletes only the prefix (`SCAN` + `DEL`) and refuses an empty one |
| Sessions | `rc.SessionStore(prefix)` implements `session.Store` |
| Queue | `rc.Queue(redis.QueueOptions{Prefix, Grace})`: sorted sets under `{queue}:` (one Cluster slot), atomic Lua reservation, failed jobs for `queue:failed`, `queue:retry`, `queue:forget` |
| Rate limiting | `rc.RateLimiter("api:", 60, time.Minute)` implements `ratelimit.Limiter` with GCRA, shared by every server |
| Locks | `cache.New(rc.CacheStore("locks:"), "").NewLock(name, ttl)` |
| Pub/sub | `rc.Publish(ctx, channel, b)`, `msgs, err := rc.Subscribe(ctx, channels...)` |

- Works with Redis 7+ and Valkey 8+; CI runs the `cachetest` and `queuetest` conformance suites on both.
- `Open` pings the server, so the app does not boot while Redis is down; a malformed URL always fails. `redis.OpenLazy(url)` only parses the URL and connects on first use, so the process starts and `rc.Ping` reports the outage in a readiness check (`app.AddHealthCheck`). go-redis retries a failed command 3 times with backoff, so that `Ping` takes seconds while Redis is down; for a fast readiness answer set `?dial_timeout=1s&max_retries=-1` on the URL (or use a separate client with those options for the check).
- `redis.New(client)` wraps an existing go-redis client (Cluster, Sentinel, custom options); `rc.Redis()` returns it.
